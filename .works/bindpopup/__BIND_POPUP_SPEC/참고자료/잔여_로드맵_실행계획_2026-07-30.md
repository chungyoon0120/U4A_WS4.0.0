HTML5 `_checkDragData`([designArea.js:188-198](../../../www/ws30/ws10_20/Popups/bindPopup/designArea/designArea.js))는 `RETCD="E"`만 두고 **RTMSG를 비워** 드롭 실패가 무음이다.

**Files:** Modify `www/ws30/ws10_20/Popups/bindPopup/designArea/designArea.js` (`_checkDragData` + 드롭 실패 표시부)

### Task G4-1: `_checkDragData` 4종 메시지 이식
- [ ] **Step 1: 각 실패 분기에 원본 메시지 키 세팅** — `H.z()`로 문구 부여 (H.z = ZMSG 조회 헬퍼, 이미 이 파일에서 사용 중)

```js
function _checkDragData(sPrc001) {
    var r = { RETCD: "", RTMSG: "", IF_DATA: null };
    if (!sPrc001) { r.RETCD = "E"; r.RTMSG = H.z("099"); return r; }          // 099 Drag 정보 없음
    var o;
    try { o = JSON.parse(sPrc001); } catch (e) { r.RETCD = "E"; r.RTMSG = H.z("100"); return r; }  // 100 잘못된 Drag 정보
    if (!o || o.PRCCD !== "PRC001") { r.RETCD = "E"; r.RTMSG = H.z("101"); return r; }              // 101 작업 수행 불가
    if (o.DnDRandKey !== oAPP.attr.DnDRandKey) { r.RETCD = "E"; r.RTMSG = H.z("102"); return r; }   // 102 다른 영역 Drag
    if (o.RETCD === "E") { r.RETCD = "E"; r.RTMSG = o.RTMSG || ""; return r; }  // 추가속성 검증 오류(드래그 payload에 실려온 것)
    r.IF_DATA = o.IF_DATA;
    return r;
}
```
